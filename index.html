/* BTK Akademi Sertifikalı C Programlama - Dinamik Bellek Yönetimi */
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 1024;
    int *pixel_buffer = (int*)malloc(n * sizeof(int));
    
    if (!pixel_buffer) {
        exit(1);
    }
    
    for (int i = 0; i < n; i++) {
        pixel_buffer[i] = i * i;
    }
    
    free(pixel_buffer);
    return 0;
}
